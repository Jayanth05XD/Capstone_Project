--- Q5 - 5-star books by category, priciest first (JOIN) ---
SQL:
SELECT b.title, b.rating, b.price_inr, c.category_name
        FROM books b
        JOIN categories c ON b.category_id = c.category_id
        WHERE b.rating = 5
        ORDER BY c.category_name, b.price_inr DESC
        LIMIT 15;
